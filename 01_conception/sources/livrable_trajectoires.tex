\section{Livrable 6 -- Trajectoires du robot}

\begin{UPSTIactivite}[][Trajectoires envisagées]
    \UPSTIquestion{Lister les \textbf{déplacements} que le robot SCARA devra
    réaliser pour assurer le traitement complet d'un gobelet (prise,
    trempage dans une cuve, dépose/évacuation).}

    \UPSTIquestion{Pour chaque déplacement identifié, préciser le
    \textbf{type de mouvement} envisagé (linéaire, circulaire, point à
    point, etc.) en justifiant ce choix au regard de la trajectoire
    souhaitée et des éventuels obstacles à éviter.}

    \UPSTIquestion{Proposer, pour chaque déplacement, un ordre de grandeur
    de \textbf{vitesse} adapté, en tenant compte des contraintes du procédé
    (temps de trempage minimum/maximum, fragilité éventuelle des pièces,
    sécurité des opérateurs à proximité).}
\end{UPSTIactivite}
